# F1 Break Reminder Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a local VS Code extension that shows a green/yellow/red F1-style flag in the Secondary Side Bar based on elapsed coding time in the current window, playing a sound on each color transition, resettable by clicking the flag.

**Architecture:** A CommonJS VS Code extension (no build step, no TypeScript) with an extension-host timer (`src/extension.js`) that computes elapsed minutes and zone color, and a `WebviewView` (`src/flagViewProvider.js` + `media/*`) that renders the flag, plays audio, and reports clicks back via `postMessage`.

**Tech Stack:** Node.js (bundled with VS Code's extension host), VS Code Extension API (`vscode` module, `WebviewViewProvider`), plain HTML/CSS/JS for the webview, no external npm dependencies.

## Global Constraints

- No TypeScript, no bundler, no external npm dependencies — plain CommonJS JS files only (YAGNI: personal single-user extension).
- Settings: `breakReminder.yellowThresholdMinutes` (default `60`), `breakReminder.redThresholdMinutes` (default `90`), both plain numbers.
- Sound files are synthesized WAV tones generated by a local script — never downloaded — named `media/green.wav`, `media/yellow.wav`, `media/red.wav` so they can be swapped later with user-supplied files of the same names.
- Testing: per the approved spec, this is manual verification only for VS Code-API-dependent code. Pure logic functions (`computeZone`, `formatElapsed`) are still verified immediately via one-off `node -e` commands (cheap, no test framework needed) before being wired into the extension.
- Elapsed time timer is wall-clock since window activation (no persistence across restarts), reset only by clicking the flag.
- Tick interval: 10000ms (10 seconds).

function formatElapsed(minutes) {
  const totalMinutes = Math.max(0, Math.floor(minutes));
  const hours = Math.floor(totalMinutes / 60);
  const mins = totalMinutes % 60;

### Task 4: Generate synthesized tone WAV files

- [ ] **Step 1: Write `scripts/generate-tones.js`**

```js
const fs = require('fs');
const path = require('path');

const SAMPLE_RATE = 44100;

function envelope(i, n, attack, release) {
  const attackSamples = Math.floor(attack * SAMPLE_RATE);
  const releaseSamples = Math.floor(release * SAMPLE_RATE);
  if (i < attackSamples) return i / attackSamples;
  if (i > n - releaseSamples) return Math.max(0, (n - i) / releaseSamples);
  return 1;
}

function tone(freq, durationSeconds, attack, release, amplitude) {
  const n = Math.floor(durationSeconds * SAMPLE_RATE);
  const samples = new Array(n);
  for (let i = 0; i < n; i++) {
    const t = i / SAMPLE_RATE;
    samples[i] = Math.sin(2 * Math.PI * freq * t) * amplitude * envelope(i, n, attack, release);
  }
  return samples;
}

function silence(durationSeconds) {
  const n = Math.floor(durationSeconds * SAMPLE_RATE);
  return new Array(n).fill(0);
}

function concat(...chunks) {
  return chunks.reduce((acc, c) => acc.concat(c), []);
}

function writeWav(filePath, samples) {
  const dataSize = samples.length * 2;
  const buffer = Buffer.alloc(44 + dataSize);
  buffer.write('RIFF', 0);
  buffer.writeUInt32LE(36 + dataSize, 4);
  buffer.write('WAVE', 8);
  buffer.write('fmt ', 12);
  buffer.writeUInt32LE(16, 16);
  buffer.writeUInt16LE(1, 20);
  buffer.writeUInt16LE(1, 22);
  buffer.writeUInt32LE(SAMPLE_RATE, 24);
  buffer.writeUInt32LE(SAMPLE_RATE * 2, 28);
  buffer.writeUInt16LE(2, 32);
  buffer.writeUInt16LE(16, 34);
  buffer.write('data', 36);
  buffer.writeUInt32LE(dataSize, 40);
  for (let i = 0; i < samples.length; i++) {
    const clamped = Math.max(-1, Math.min(1, samples[i]));
    buffer.writeInt16LE(Math.round(clamped * 32767), 44 + i * 2);
  }
  fs.writeFileSync(filePath, buffer);
}

const mediaDir = path.join(__dirname, '..', 'media');

// Green: single soft chime
writeWav(path.join(mediaDir, 'green.wav'), tone(880, 0.3, 0.02, 0.2, 0.4));

// Yellow: two short beeps
writeWav(
  path.join(mediaDir, 'yellow.wav'),
  concat(tone(660, 0.15, 0.01, 0.05, 0.5), silence(0.08), tone(660, 0.15, 0.01, 0.05, 0.5))
);

// Red: urgent alternating alarm, repeated 3 times
writeWav(
  path.join(mediaDir, 'red.wav'),
  concat(
    tone(880, 0.15, 0.005, 0.03, 0.6),
    tone(440, 0.15, 0.005, 0.03, 0.6),
    tone(880, 0.15, 0.005, 0.03, 0.6),
    tone(440, 0.15, 0.005, 0.03, 0.6),
    tone(880, 0.15, 0.005, 0.03, 0.6),
    tone(440, 0.15, 0.005, 0.03, 0.6)
  )
);

console.log('Generated green.wav, yellow.wav, red.wav in', mediaDir);
```

```js
(function () {
  const vscode = acquireVsCodeApi();
  const flagEl = document.getElementById('flag');
  const elapsedEl = document.getElementById('elapsed');
  const audioEls = {
    green: document.getElementById('audio-green'),
    yellow: document.getElementById('audio-yellow'),
    red: document.getElementById('audio-red'),
  };

  let lastColor = 'green';

  function setColor(color) {
    flagEl.classList.remove('flag-green', 'flag-yellow', 'flag-red');
    flagEl.classList.add('flag-' + color);
  }

  window.addEventListener('message', (event) => {
    const message = event.data;
    if (message.type === 'tick') {
      elapsedEl.textContent = formatElapsed(message.elapsedMinutes);
      if (message.color !== lastColor) {
        setColor(message.color);
        const audioEl = audioEls[message.color];
        if (audioEl) {
          audioEl.currentTime = 0;
          audioEl.play().catch(() => {});
        }
        lastColor = message.color;
      }
    }
  });

  flagEl.addEventListener('click', () => {
    setColor('green');
    elapsedEl.textContent = formatElapsed(0);
    lastColor = 'green';
    vscode.postMessage({ type: 'reset' });
  });
})();
```

### Task 7: FlagViewProvider

**Files:**
- Create: `src/flagViewProvider.js`

**Interfaces:**
- Consumes: `media/webview.html`, `media/webview.css`, `media/webview.js`, `media/format.js`, `media/green.wav`, `media/yellow.wav`, `media/red.wav` (all from earlier tasks), and VS Code's `vscode.window.registerWebviewViewProvider` contract.
- Produces: `class FlagViewProvider` with:
  - `constructor(extensionUri)`
  - `resolveWebviewView(webviewView)` — satisfies `vscode.WebviewViewProvider`
  - `onReset(callback)` — registers a callback invoked when the webview posts `{ type: 'reset' }`
  - `postTick(color, elapsedMinutes)` — sends `{ type: 'tick', color, elapsedMinutes }` to the webview if it's currently resolved
  - Used by `src/extension.js` in Task 8.

- [ ] **Step 1: Write `src/flagViewProvider.js`**

```js
const vscode = require('vscode');
const fs = require('fs');

class FlagViewProvider {
  constructor(extensionUri) {
    this._extensionUri = extensionUri;
    this._view = undefined;
    this._onReset = undefined;
  }

  resolveWebviewView(webviewView) {
    this._view = webviewView;
    webviewView.webview.options = {
      enableScripts: true,
      localResourceRoots: [vscode.Uri.joinPath(this._extensionUri, 'media')],
    };
    webviewView.webview.html = this._buildHtml(webviewView.webview);
    webviewView.webview.onDidReceiveMessage((message) => {
      if (message && message.type === 'reset' && this._onReset) {
        this._onReset();
      }
    });
  }

  onReset(callback) {
    this._onReset = callback;
  }

  postTick(color, elapsedMinutes) {
    if (this._view) {
      this._view.webview.postMessage({ type: 'tick', color, elapsedMinutes });
    }
  }

  _buildHtml(webview) {
    const mediaUri = (file) =>
      webview.asWebviewUri(vscode.Uri.joinPath(this._extensionUri, 'media', file)).toString();

    const htmlPath = vscode.Uri.joinPath(this._extensionUri, 'media', 'webview.html').fsPath;
    let html = fs.readFileSync(htmlPath, 'utf8');
    html = html
      .replace('{{cssUri}}', mediaUri('webview.css'))
      .replace('{{formatUri}}', mediaUri('format.js'))
      .replace('{{jsUri}}', mediaUri('webview.js'))
      .replace('{{greenUri}}', mediaUri('green.wav'))
      .replace('{{yellowUri}}', mediaUri('yellow.wav'))
      .replace('{{redUri}}', mediaUri('red.wav'));
    return html;
  }
}

module.exports = { FlagViewProvider };
```

### Task 8: Wire up extension activation

**Files:**
- Modify: `src/extension.js` (replace the Task 1 stub entirely)

**Interfaces:**
- Consumes: `computeZone` from `src/zone.js` (Task 2), `FlagViewProvider` from `src/flagViewProvider.js` (Task 7), and reads `breakReminder.yellowThresholdMinutes` / `breakReminder.redThresholdMinutes` from VS Code configuration (Task 1).

- [ ] **Step 1: Replace `src/extension.js`**

```js
const vscode = require('vscode');
const { computeZone } = require('./zone');
const { FlagViewProvider } = require('./flagViewProvider');

const TICK_INTERVAL_MS = 10000;

function activate(context) {
  const provider = new FlagViewProvider(context.extensionUri);
  context.subscriptions.push(
    vscode.window.registerWebviewViewProvider('f1BreakReminder.flagView', provider)
  );

  let startTime = Date.now();

  provider.onReset(() => {
    startTime = Date.now();
  });

  const interval = setInterval(() => {
    const elapsedMinutes = (Date.now() - startTime) / 60000;
    const config = vscode.workspace.getConfiguration('breakReminder');
    const yellowThreshold = config.get('yellowThresholdMinutes', 60);
    const redThreshold = config.get('redThresholdMinutes', 90);
    const color = computeZone(elapsedMinutes, yellowThreshold, redThreshold);
    provider.postTick(color, elapsedMinutes);
  }, TICK_INTERVAL_MS);

  context.subscriptions.push({ dispose: () => clearInterval(interval) });
}

function deactivate() {}

module.exports = { activate, deactivate };
```
